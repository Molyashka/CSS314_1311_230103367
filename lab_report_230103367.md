### Cache-line boundary identification and Karp-Flatt diagnosis

The largest consecutive drop in Table 1A is 8 B → 16 B (1.02x), which is well within measurement noise (IQR up to 237 ms). **No cache-line boundary inflection was observed in this experiment**, so I cannot infer the coherence line width from these data alone (the hardware value is reported in Section 1).

The Karp-Flatt trajectories for stride = 1 and stride = 16 are nearly identical. Both start near 0.01-0.02 at p = 2, and both jump between p = 3 and p = 4 (0.035 → 0.111 and 0.031 → 0.099) and then rise slowly to 0.160 and 0.144 at p = 8. If false sharing were the dominant overhead, e(p) under stride = 1 would rise sharply and stay far above the stride = 16 curve. It does not: the gap is small (0.160 vs 0.144 at p = 8). The likely explanation is that each loop iteration in CPython costs far more (hundreds of nanoseconds of interpreter work per update) than the cost of bouncing a cache line, so coherence traffic is hidden. The shared rise of e(p) beyond p = 4 in both series points to a different cause, most likely heterogeneous cores / OS scheduling and shared-resource limits rather than cache-line contention. Therefore the expected "e(p) rises sharply under stride = 1" was not reproduced on this machine.

### Analysis

- **Block static** is worst (λ = 2.25): the last worker receives the heaviest items (cost grows quadratically with i) while the others idle at the join, making it about 1.9x slower than the best schedule.
- **Cyclic** spreads heavy items across workers and is almost perfectly balanced (λ = 1.005).
- **Dynamic chunks** show the trade-off: at C = 1, balance is perfect (λ = 1.000) but each item needs a lock acquisition on the shared counter, so the time is higher (8.8 s). As C grows, synchronization falls but tail imbalance grows (λ = 1.02, 1.07, 1.17 for C = 64, 256, 1024), and the time rises again at C = 1024 (8.7 s). The minimum is at **C\* = 64** (7.67 s); C = 16 is statistically indistinguishable (7.71 s).
- Dynamic scheduling at C\* is about 1.9x faster than block static and about 7% faster than cyclic, though the cyclic-vs-dynamic gap is comparable to the IQR.

```python
from dataclasses import dataclass
import multiprocessing as mp
import time
from typing import Callable


@dataclass
class BenchStats:
    min_ms: float
    median_ms: float
    iqr_ms: float
    max_ms: float


def fnv1a_hash(s: str) -> int:
    h = 14695981039346656037
    for b in s.encode("utf-8"):
        h ^= b
        h = (h * 1099511628211) & 0xFFFFFFFFFFFFFFFF
    return h


def run_benchmark(fn: Callable[[], None], warmup_runs: int = 3, timed_runs: int = 9) -> BenchStats:
    for _ in range(warmup_runs):
        fn()
    samples = []
    for _ in range(timed_runs):
        t0 = time.perf_counter_ns()
        fn()
        t1 = time.perf_counter_ns()
        samples.append((t1 - t0) / 1e6)
    samples.sort()
    mid = timed_runs // 2
    med = samples[mid] if timed_runs % 2 == 1 else 0.5 * (samples[mid - 1] + samples[mid])
    q25 = samples[timed_runs // 4]
    q75 = samples[(3 * timed_runs) // 4]
    return BenchStats(samples[0], med, q75 - q25, samples[-1])


def emit_verify_tag(task_id: str, student_seed: int,
                    primary_metric: float, secondary_metric: float) -> str:
    hw = mp.cpu_count()
    payload = (
        f"{task_id}|{student_seed}|{hw}|"
        f"{primary_metric:.4f}|{secondary_metric:.4f}"
    )
    sig = fnv1a_hash(payload)
    return f"VERIFY_HASH[{task_id}]: {sig:016X} (P_log={hw})"
```

```python
"""Phase 1: false sharing stride sweep + Karp-Flatt series."""
import ctypes
import multiprocessing as mp
from multiprocessing import shared_memory

from harness import run_benchmark, emit_verify_tag
from lab_config import STUDENT_ID, P_PHYS, P_LOG, QUICK, md_table

M = 24_000_000 // (20 if QUICK else 1)
WARM, TIMED = (1, 3) if QUICK else (2, 7)
STRIDES = [1, 2, 4, 8, 16, 32]


def fs_worker(shm_name: str, slot_idx: int, iters: int) -> None:
    shm = shared_memory.SharedMemory(name=shm_name)
    # length derived from the buffer so large p*stride never indexes out of range
    arr = (ctypes.c_uint64 * (shm.size // 8)).from_buffer(shm.buf)
    acc = arr[slot_idx]
    mask = 0xFFFFFFFFFFFFFFFF
    mult = 6364136223846793005
    for i in range(iters):
        acc = (acc * mult + (i | 1)) & mask
        arr[slot_idx] = acc          # direct hardware store to shared cache line
    del arr                          # release buffer export before closing SharedMemory
    shm.close()


def run_once(shm_name, p, stride):
    iters = M // p
    procs = [mp.Process(target=fs_worker, args=(shm_name, t * stride, iters)) for t in range(p)]
    for pr in procs:
        pr.start()
    for pr in procs:
        pr.join()


def measure(p, stride):
    nslots = max(512, p * stride + 1)
    shm = shared_memory.SharedMemory(create=True, size=nslots * 8)
    try:
        return run_benchmark(lambda: run_once(shm.name, p, stride),
                             warmup_runs=WARM, timed_runs=TIMED)
    finally:
        shm.close()
        shm.unlink()


def karp_flatt(S, p):
    return (1.0 / S - 1.0 / p) / (1.0 - 1.0 / p)


def main():
    print(f"# Phase 1 (P_phys={P_PHYS}, P_log={P_LOG}, M={M:,})\n")

    # ---- Table 1A: stride sweep at p = P_phys
    res = {}
    for s in STRIDES:
        res[s] = measure(P_PHYS, s)
        print(f"  [stride {s} done: {res[s].median_ms:.1f} ms]", flush=True)
    rows = []
    for s in STRIDES:
        st = res[s]
        mops = M / (st.median_ms / 1e3) / 1e6
        rows.append([s, s * 8, f"{st.median_ms:.2f}", f"{st.iqr_ms:.2f}", f"{mops:.2f}"])
    print("\n## Table 1A - Stride sensitivity at p = P_phys\n")
    print(md_table(["stride (words)", "separation (bytes)", "median (ms)", "IQR (ms)", "Mops/s"], rows))

    # cache-line boundary: largest relative drop between consecutive strides
    best = max(range(len(STRIDES) - 1),
               key=lambda i: res[STRIDES[i]].median_ms / res[STRIDES[i + 1]].median_ms)
    ratio = res[STRIDES[best]].median_ms / res[STRIDES[best + 1]].median_ms
    print(f"\nLargest drop: {STRIDES[best]*8} B -> {STRIDES[best+1]*8} B "
          f"({ratio:.2f}x faster). Inferred coherence line width ~ {STRIDES[best+1]*8} bytes.")

    # ---- Table 1B: Karp-Flatt for stride 1 and 16
    ps = list(range(1, P_LOG + 1))
    series = {}
    for stride in (1, 16):
        T = {}
        for p in ps:
            T[p] = measure(p, stride).median_ms
            print(f"  [stride {stride}, p={p}: {T[p]:.1f} ms]", flush=True)
        series[stride] = T
        rows = []
        for p in ps:
            S = T[1] / T[p]
            E = S / p
            e = f"{karp_flatt(S, p):.4f}" if p >= 2 else "-"
            rows.append([p, f"{T[p]:.2f}", f"{S:.3f}", f"{E:.3f}", e])
        print(f"\n## Table 1B - stride = {stride} ({stride*8} B separation)\n")
        print(md_table(["p", "T_p (ms)", "S_p", "E_p", "e(p)"], rows))

    t1 = res[1].median_ms
    t16 = res[16].median_ms
    print(f"\nmedian_ms(stride=1)={t1:.4f}  median_ms(stride=16)={t16:.4f}  at p={P_PHYS}")
    print(emit_verify_tag("PHASE1", STUDENT_ID, t1, t16))
```

```python
"""Phase 2: strong scaling (Amdahl) and weak scaling (Gustafson)."""
import math
import multiprocessing as mp

from harness import run_benchmark, emit_verify_tag
from lab_config import STUDENT_ID, P_PHYS, P_LOG, QUICK, md_table

F = 20 if QUICK else 1
N_STRONG = 1_600_000 // F
N0 = 250_000 // F
WARM, TIMED = (1, 3) if QUICK else (1, 5)


def stage_a(n_serial: int) -> float:
    """Serial bottleneck: 12 transcendental ops (6 x sin+sqrt) per element."""
    acc = 0.0
    for i in range(n_serial):
        x = i * 1e-3 + 0.5
        for _ in range(6):
            x = math.sqrt(abs(math.sin(x)) + 1.0)
        acc += x
    return acc


def stage_b_kernel(rng):
    """Parallel kernel: 35 multiply-add + sin iterations per element, no shared writes."""
    start, end = rng
    acc = 0.0
    for i in range(start, end):
        x = (i % 1000) * 1e-3
        for _ in range(35):
            x = math.sin(x) * 0.99 + 0.05
        acc += x
    return acc


def split(n, p):
    return [(n * t // p, n * (t + 1) // p) for t in range(p)]


def time_trial(N, p):
    """Median total (Stage A + Stage B) with a persistent Pool created OUTSIDE the timed region."""
    with mp.Pool(processes=p) as pool:
        chunks = split(N, p)
        n_serial = int(0.05 * N)

        def trial():
            stage_a(n_serial)
            pool.map(stage_b_kernel, chunks, chunksize=1)

        return run_benchmark(trial, warmup_runs=WARM, timed_runs=TIMED).median_ms


def time_stage_a(N):
    n_serial = int(0.05 * N)
    return run_benchmark(lambda: stage_a(n_serial), warmup_runs=WARM, timed_runs=TIMED).median_ms


def main():
    ps = list(range(1, P_LOG + 1))
    print(f"# Phase 2 (P_phys={P_PHYS}, P_log={P_LOG}, N_strong={N_STRONG:,}, N0={N0:,})\n")

    # ---- 2A strong
    Ts = {}
    for p in ps:
        Ts[p] = time_trial(N_STRONG, p)
        print(f"  [strong p={p}: {Ts[p]:.1f} ms]", flush=True)
    Ss = {p: Ts[1] / Ts[p] for p in ps}
    rows = [[p, f"{Ts[p]:.2f}", f"{Ss[p]:.3f}", f"{Ss[p]/p:.3f}"] for p in ps]
    print("\n## Experiment 2A - Strong scaling\n")
    print(md_table(["p", "T_strong (ms)", "S_strong", "E_strong"], rows))

    # ---- 2B weak
    Tw, TA = {}, {}
    for p in ps:
        N = p * N0
        Tw[p] = time_trial(N, p)
        TA[p] = time_stage_a(N)
        print(f"  [weak p={p}: {Tw[p]:.1f} ms]", flush=True)
    Gs = {}
    rows = []
    for p in ps:
        alpha = TA[p] / Tw[p]
        Gs[p] = p - alpha * (p - 1)
        rows.append([p, p * N0, f"{Tw[p]:.2f}", f"{Tw[1]/Tw[p]:.3f}", f"{alpha:.4f}", f"{Gs[p]:.3f}"])
    print("\n## Experiment 2B - Weak scaling\n")
    print(md_table(["p", "N(p)", "T_weak (ms)", "E_weak", "alpha", "S_Gustafson"], rows))

    # ---- 3.3 analysis
    TA1 = time_stage_a(N_STRONG)
    s = TA1 / Ts[1]
    s_inf = 1.0 / s
    s_amdahl_plog = 1.0 / (s + (1 - s) / P_LOG)
    print("\n## Analytical deliverables\n")
    print(f"- Serial fraction s = T_StageA(1)/T1_strong = {TA1:.3f}/{Ts[1]:.3f} = {s:.5f}")
    print(f"- Amdahl S_inf = {s_inf:.3f}; theoretical S at p=P_log={P_LOG}: {s_amdahl_plog:.3f}; "
          f"observed S_strong(P_log) = {Ss[P_LOG]:.3f}")

    if P_LOG > P_PHYS and P_PHYS >= 2:
        before = Ss[P_PHYS] - Ss[P_PHYS - 1]
        after = Ss[P_PHYS + 1] - Ss[P_PHYS]
        drop = (1 - after / before) * 100 if before > 0 else float("nan")
        print(f"- Marginal speedup: p={P_PHYS-1}->{P_PHYS}: {before:.3f}; "
              f"p={P_PHYS}->{P_PHYS+1}: {after:.3f}; drop = {drop:.1f}%")
    else:
        print("- No SMT / P_log == P_phys: boundary analysis not applicable "
              "(or set P_PHYS correctly in lab_config.py).")

    print(f"\nS_strong(P_log)={Ss[P_LOG]:.4f}  S_Gustafson(P_log)={Gs[P_LOG]:.4f}")
    print(emit_verify_tag("PHASE2", STUDENT_ID, Ss[P_LOG], Gs[P_LOG]))


if __name__ == "__main__":
    main()
```

```python
"""Phase 3A: sustained DRAM bandwidth, Python compute ceiling, ridge point."""
import json
import multiprocessing as mp
from multiprocessing import shared_memory

from harness import run_benchmark
from lab_config import P_PHYS, QUICK, attach_shm, md_table

SIZE = (96 * 10**6) // (10 if QUICK else 1)       # bytes per buffer
FLOPS_PER_WORKER = 8.0e7 / (10 if QUICK else 1)
STMTS = 16                                         # unrolled statements per loop iteration
WARM, TIMED = (1, 3) if QUICK else (2, 7)


def copy_worker(args):
    name_a, name_b, s, e = args
    a = attach_shm(name_a)
    b = attach_shm(name_b)
    try:
        a.buf[s:e] = b.buf[s:e]          # slice copy -> native memcpy
    finally:
        a.close()
        b.close()


def flop_worker(iters):
    a, b = 1.0000001, 1e-9
    x0 = x1 = x2 = x3 = x4 = x5 = x6 = x7 = 0.5
    x8 = x9 = x10 = x11 = x12 = x13 = x14 = x15 = 0.25
    for _ in range(iters):
        x0 = x0 * a + b
        x1 = x1 * a + b
        x2 = x2 * a + b
        x3 = x3 * a + b
        x4 = x4 * a + b
        x5 = x5 * a + b
        x6 = x6 * a + b
        x7 = x7 * a + b
        x8 = x8 * a + b
        x9 = x9 * a + b
        x10 = x10 * a + b
        x11 = x11 * a + b
        x12 = x12 * a + b
        x13 = x13 * a + b
        x14 = x14 * a + b
        x15 = x15 * a + b
    return x0 + x1 + x2 + x3 + x4 + x5 + x6 + x7 + x8 + x9 + x10 + x11 + x12 + x13 + x14 + x15


def main():
    p = P_PHYS
    A = shared_memory.SharedMemory(create=True, size=SIZE)
    B = shared_memory.SharedMemory(create=True, size=SIZE)
    try:
        A.buf[:SIZE] = b"\x01" * SIZE        # touch pages so we measure real DRAM, not zero-page tricks
        B.buf[:SIZE] = b"\x02" * SIZE
        tasks = [(A.name, B.name, SIZE * t // p, SIZE * (t + 1) // p) for t in range(p)]
        iters = int(FLOPS_PER_WORKER // STMTS)

        with mp.Pool(processes=p) as pool:
            st = run_benchmark(lambda: pool.map(copy_worker, tasks, chunksize=1),
                               warmup_runs=WARM, timed_runs=TIMED)
            bw = 2 * SIZE / ((st.median_ms / 1e3) * 1e9)

            sc = run_benchmark(lambda: pool.map(flop_worker, [iters] * p, chunksize=1),
                               warmup_runs=WARM, timed_runs=TIMED)
            psus = (iters * STMTS * p) / ((sc.median_ms / 1e3) * 1e9)
    finally:
        A.close(); A.unlink()
        B.close(); B.unlink()

    ridge = psus / bw
    print(f"# Phase 3A (P_phys={p})\n")
    print(md_table(["Quantity", "Value"], [
        ["Copy median time (ms)", f"{st.median_ms:.3f} (IQR {st.iqr_ms:.3f})"],
        ["BW_sustained (GB/s)", f"{bw:.3f}"],
        ["Compute median time (ms)", f"{sc.median_ms:.3f} (IQR {sc.iqr_ms:.3f})"],
        ["P_sustained (GFLOP/s)", f"{psus:.4f}"],
        ["Ridge point I* (FLOP/Byte)", f"{ridge:.5f}"],
    ]))
    print("\n(Each `x = x*a + b` statement is counted as 1 FMA FLOP, per the handout wording.)")
    with open("roofline.json", "w") as f:
        json.dump({"bw": bw, "psus": psus, "ridge": ridge}, f)
    print("Saved roofline.json (used by task3b_gemm.py)")


if __name__ == "__main__":
    main()
```

```python
"""Phase 3B: 3 parallel GEMM variants, 10-point tile sweep, roofline mapping."""
import json
import os
import multiprocessing as mp
from array import array
from multiprocessing import shared_memory

from harness import run_benchmark, emit_verify_tag
from lab_config import (STUDENT_ID, P_PHYS, L1D_KB, L2_KB, QUICK,
                        attach_shm, md_table)

N = 96 if QUICK else 384
TILES = [4, 8, 12, 16, 24, 32, 48, 64, 96, 128]
WARM, TIMED = (1, 3) if QUICK else (1, 5)


# ------------------------- kernels (module level for spawn) -------------------------
def _v1(A, B, C, r0, r1):
    n = N
    for i in range(r0, r1):
        for j in range(n):
            dot = 0.0
            for k in range(n):
                dot += A[i * n + k] * B[k * n + j]
            C[i * n + j] = dot


def _v2(A, B, C, r0, r1):
    n = N
    C[r0 * n:r1 * n] = array("d", bytes(8 * (r1 - r0) * n))
    for i in range(r0, r1):
        iN = i * n
        for k in range(n):
            a_ik = A[iN + k]
            kN = k * n
            for j in range(n):
                C[iN + j] += a_ik * B[kN + j]


def _v3(A, B, C, r0, r1, bs):
    n = N
    C[r0 * n:r1 * n] = array("d", bytes(8 * (r1 - r0) * n))
    for ii in range(r0, r1, bs):
        i_end = min(ii + bs, r1)
        for jj in range(0, n, bs):
            j_end = min(jj + bs, n)
            for kk in range(0, n, bs):
                k_end = min(kk + bs, n)
                for i in range(ii, i_end):
                    iN = i * n
                    base = iN + jj
                    for k in range(kk, k_end):
                        a_ik = A[iN + k]                       # hoisted out of j loop
                        brow = B[k * n + jj:k * n + j_end]     # hoisted tile slice
                        for jo, bv in enumerate(brow):
                            C[base + jo] += a_ik * bv


def gemm_worker(task):
    variant, r0, r1, bs, names = task
    shms = [attach_shm(nm) for nm in names]
    A, B, C = [s.buf.cast("d") for s in shms]
    try:
        if variant == 1:
            _v1(A, B, C, r0, r1)
        elif variant == 2:
            _v2(A, B, C, r0, r1)
        else:
            _v3(A, B, C, r0, r1, bs)
    finally:
        for v in (A, B, C):
            v.release()
        for s in shms:
            s.close()


# ------------------------- driver -------------------------
def main():
    p = P_PHYS
    nbytes = 8 * N * N
    shms = [shared_memory.SharedMemory(create=True, size=nbytes) for _ in range(3)]
    names = [s.name for s in shms]
    try:
        A = shms[0].buf.cast("d")
        B = shms[1].buf.cast("d")
        A[:N * N] = array("d", [((i * 37) % 101) / 101.0 for i in range(N * N)])
        B[:N * N] = array("d", [((i * 53) % 103) / 103.0 for i in range(N * N)])
        A.release(); B.release()

        def snapshot_C():
            a = array("d")
            a.frombytes(bytes(shms[2].buf[:nbytes]))
            return a

        def bench(pool, variant, bs=0):
            tasks = [(variant, N * t // p, N * (t + 1) // p, bs, names) for t in range(p)]
            return run_benchmark(lambda: pool.map(gemm_worker, tasks, chunksize=1),
                                 warmup_runs=WARM, timed_runs=TIMED)

        gflops = lambda ms: 2 * N ** 3 / ((ms / 1e3) * 1e9)

        with mp.Pool(processes=p) as pool:
            s1 = bench(pool, 1)
            print(f"  [V1 {s1.median_ms:.0f} ms]", flush=True)
            s2 = bench(pool, 2)
            c2 = snapshot_C()
            print(f"  [V2 {s2.median_ms:.0f} ms]", flush=True)
            s3, maxdiff = {}, {}
            for bs in TILES:
                s3[bs] = bench(pool, 3, bs)
                c3 = snapshot_C()
                maxdiff[bs] = max(abs(x - y) for x, y in zip(c3, c2))
                print(f"  [V3 BS={bs}: {s3[bs].median_ms:.0f} ms]", flush=True)
    finally:
        for s in shms:
            s.close()
            s.unlink()

    g1, g2 = gflops(s1.median_ms), gflops(s2.median_ms)
    g3 = {bs: gflops(s3[bs].median_ms) for bs in TILES}
    best = max(TILES, key=lambda b: g3[b])

    print(f"\n# Phase 3B (N={N}, P_phys={p})\n")
    print(md_table(["Variant", "median (ms)", "IQR (ms)", "GFLOP/s"], [
        ["1 naive i-j-k", f"{s1.median_ms:.2f}", f"{s1.iqr_ms:.2f}", f"{g1:.5f}"],
        ["2 i-k-j", f"{s2.median_ms:.2f}", f"{s2.iqr_ms:.2f}", f"{g2:.5f}"],
    ]))

    roof = None
    if os.path.exists("roofline.json"):
        with open("roofline.json") as f:
            roof = json.load(f)
    attain = lambda I: min(roof["psus"], roof["bw"] * I) if roof else None
    pct = lambda g, I: f"{100 * g / attain(I):.2f}%" if roof else "n/a (run task3a first)"

    rows = []
    for bs in TILES:
        w = 3 * bs * bs * 8
        fits = "L1d" if w <= L1D_KB * 1024 else ("L2" if w <= L2_KB * 1024 else ">L2")
        rows.append([bs, f"{s3[bs].median_ms:.2f}", f"{s3[bs].iqr_ms:.2f}", f"{g3[bs]:.5f}",
                     w, f"{w/1024:.1f}", fits, f"{bs/12:.3f}", pct(g3[bs], bs / 12),
                     f"{maxdiff[bs]:.1e}"])
    print("\n## Variant 3 tile sweep\n")
    print(md_table(["BS", "median (ms)", "IQR (ms)", "GFLOP/s", "W_tile (B)", "W_tile (KB)",
                    "fits in", "I_tiled", "% of roofline", "max|C3-C2|"], rows))
    print(f"\nBS* = {best} (peak {g3[best]:.5f} GFLOP/s). "
          f"Numerical check passes (<1e-9) for all BS: {all(d < 1e-9 for d in maxdiff.values())}")

    print("\n## Roofline mapping\n")
    print(md_table(["Variant", "I (FLOP/B)", "GFLOP/s", "% of attainable"], [

        [f"3 (BS*={best})", f"{best/12:.3f}", f"{g3[best]:.5f}", pct(g3[best], best / 12)],

if __name__ == "__main__":
```

```python
"""Phase 4: static block vs cyclic vs shared-memory dynamic chunk scheduling."""
import math
import multiprocessing as mp
import time

from harness import run_benchmark, emit_verify_tag
from lab_config import STUDENT_ID, P_PHYS, QUICK, md_table, median

N = 12_000 // (4 if QUICK else 1)
SCALE_DIV = 3600 // (16 if QUICK else 1)     # keeps K(N-1)/K(0) ratio high in quick mode
CHUNKS = [1, 4, 16, 64, 256, 1024]
WARM, TIMED = (1, 3)


def do_item(i):
    k = 1 + (i * i) // SCALE_DIV             # K(i) = 1 + floor((i/60)^2)
    x = 0.5 + i * 1e-4
    acc = 0.0
    for _ in range(k):
        s = math.sin(x)
        c = math.cos(x)
        x = math.sqrt(1.0 + abs(s * c))
        acc += s * c
    return acc


def block_worker(t, p, tw):
    t0 = time.perf_counter()
    for i in range(N * t // p, N * (t + 1) // p):
        do_item(i)
    tw[t] = (time.perf_counter() - t0) * 1e3


def cyclic_worker(t, p, tw):
    t0 = time.perf_counter()
    for i in range(t, N, p):
        do_item(i)
    tw[t] = (time.perf_counter() - t0) * 1e3


def dynamic_worker(t, p, C, counter, lock, tw):
    t0 = time.perf_counter()
    while True:
        with lock:
            start = counter.value
            counter.value = start + C
        if start >= N:
            break
        for i in range(start, min(start + C, N)):
            do_item(i)
    tw[t] = (time.perf_counter() - t0) * 1e3


def run_strategy(kind, C=0):
    """Returns (BenchStats, median lambda over timed runs)."""
    p = P_PHYS
    lams = []

    def once():
        tw = mp.Array("d", p, lock=False)
        if kind == "block":
            procs = [mp.Process(target=block_worker, args=(t, p, tw)) for t in range(p)]
        elif kind == "cyclic":
            procs = [mp.Process(target=cyclic_worker, args=(t, p, tw)) for t in range(p)]
        else:
            counter = mp.RawValue("i", 0)
            lock = mp.Lock()
            procs = [mp.Process(target=dynamic_worker, args=(t, p, C, counter, lock, tw))
                     for t in range(p)]
        for pr in procs:
            pr.start()
        for pr in procs:
            pr.join()
        ts = list(tw)
        lams.append(max(ts) / (sum(ts) / p))

    st = run_benchmark(once, warmup_runs=WARM, timed_runs=TIMED)
    return st, median(lams[-TIMED:])


def main():
    print(f"# Phase 4 (P_phys={P_PHYS}, N={N:,}, cost ratio item(N-1)/item(0) = "
          f"{1 + ((N-1)**2)//SCALE_DIV}x)\n")
    s1, l1 = run_strategy("block")
    print(f"  [block {s1.median_ms:.0f} ms]", flush=True)
    s2, l2 = run_strategy("cyclic")
    print(f"  [cyclic {s2.median_ms:.0f} ms]", flush=True)
    dyn = {}
    for C in CHUNKS:
        dyn[C] = run_strategy("dynamic", C)
        print(f"  [dynamic C={C}: {dyn[C][0].median_ms:.0f} ms]", flush=True)

    rows = [["1 Block static", "-", f"{s1.median_ms:.2f}", f"{s1.iqr_ms:.2f}", f"{l1:.3f}"],
            ["2 Cyclic static", "-", f"{s2.median_ms:.2f}", f"{s2.iqr_ms:.2f}", f"{l2:.3f}"]]
    for C in CHUNKS:
        st, lam = dyn[C]
        rows.append(["3 Dynamic chunk", C, f"{st.median_ms:.2f}", f"{st.iqr_ms:.2f}", f"{lam:.3f}"])
    print("\n## Scheduling results\n")
    print(md_table(["Strategy", "C", "T_wall median (ms)", "IQR (ms)", "lambda"], rows))

    cstar = min(CHUNKS, key=lambda c: dyn[c][0].median_ms)
    print(f"\nC* = {cstar} (T_wall = {dyn[cstar][0].median_ms:.2f} ms)")
    t_s1, t_s3 = s1.median_ms, dyn[cstar][0].median_ms
    print(f"T_wall(S1)={t_s1:.4f}  T_wall(S3,C*)={t_s3:.4f}")
    print(emit_verify_tag("PHASE4", STUDENT_ID, t_s1, t_s3))


if __name__ == "__main__":
    main()
```
